def get_user(uid):
    with sqlite3.connect(DB_NAME) as c:
        c.row_factory = sqlite3.Row
        return c.execute("SELECT * FROM users WHERE id=?", (uid,)).fetchone()

    try:
        info = api.users.get(user_ids=uid)
        if info:

def extract_id(raw_str):
    if not raw_str: return None
    match = re.search(r'\[id(\d+)\|', raw_str)
    if match: return int(match.group(1))
    nums = re.findall(r'^\d+$', raw_str.strip())
    if nums: return int(nums[0])
    return None

def resolve_user_id(api, raw_str):
    if not raw_str: return None
    # Упоминание вида [id123|Имя]
    match = re.search(r'\[id(\d+)\|', raw_str)
    if match: return int(match.group(1))
    # Чистый числовой ID
    if raw_str.strip().lstrip('-').isdigit():
        return int(raw_str.strip())
    # @username или просто username
    screen_name = raw_str.strip().lstrip('@')
    try:
        info = api.users.get(user_ids=screen_name)
        if info:
            return info[0]['id']
    except Exception as e:
        logger.error(f"Ошибка resolve_user_id({screen_name}): {e}")
    return None

def fmt_big(amount):
    if amount is None: return "0"
    if amount >= 1000000:
        return f"{amount/1000000:.1f}M"
    elif amount >= 1000:
        return f"{int(amount)}" if amount == int(amount) else f"{amount:.2f}"
    return f"{int(amount)}" if amount == int(amount) else f"{amount:.2f}".rstrip('0').rstrip('.')

def fmt_time(td):
    total_seconds = int(td.total_seconds())
    if total_seconds < 60:
        return f"{total_seconds} сек"
    elif total_seconds < 3600:
        minutes = total_seconds // 60
        seconds = total_seconds % 60
        return f"{minutes} мин {seconds} сек"
    else:
        hours = total_seconds // 3600
        minutes = (total_seconds % 3600) // 60
        return f"{hours} ч {minutes} мин"

def can_get_bonus(uid):
    user = get_user(uid)
    if not user or not user['last_bonus']:
        return True, None
    last = datetime.strptime(user['last_bonus'], "%Y-%m-%d %H:%M:%S")
    delta = timedelta(hours=3)
    if datetime.now() - last >= delta:
        return True, None
    else:
        next_time = last + delta
        remaining = next_time - datetime.now()
        return False, remaining

def handle_message_event(api, event):
    try:
        obj = event.object
        logger.info(f"MESSAGE_EVENT raw: {dict(obj)}")
    except Exception as e:
        logger.error(f"MESSAGE_EVENT dump error: {e}")
        obj = event.object

    uid = obj.get('user_id')
    peer_id = obj.get('peer_id')
    event_id = obj.get('event_id')
    payload = obj.get('payload', {})

    logger.info(f"MESSAGE_EVENT uid={uid} peer={peer_id} payload={payload!r}")

    if isinstance(payload, str):
        try:
            payload = json.loads(payload)
        except:
            payload = {}
    elif not isinstance(payload, dict):
        payload = {}

    def answer(text="✅"):
        try:
            api.messages.sendMessageEventAnswer(
                event_id=event_id,
                user_id=uid,
                peer_id=peer_id,
                event_data=json.dumps({"type": "show_snackbar", "text": text}, ensure_ascii=False)
            )
            logger.info(f"answer() sent snackbar: {text}")
        except Exception as e:
            logger.error(f"sendMessageEventAnswer error: {e}")

    try:
        cmd = payload.get('cmd')
        logger.info(f"handle_message_event cmd={cmd}")

        if cmd == 'property':
            cars = get_user_cars(uid)
            logger.info(f"property: cars={cars}")
            if not cars:
                answer("У вас пока нет машин")
                send(api, uid, "🏠 ВАШЕ ИМУЩЕСТВО 🏠\n\nУ вас пока нет машин.\nКупить можно через /машины", peer_id)
                return
            keyboard = VkKeyboard(inline=True)
            keyboard.add_callback_button("🚗 Мои машины", color=VkKeyboardColor.PRIMARY, payload={"cmd": "show_cars"})
            answer(f"Машин: {len(cars)} шт.")
            send(api, uid, f"🏠 ВАШЕ ИМУЩЕСТВО 🏠\n\n🚗 Автомобилей: {len(cars)} шт.", peer_id, keyboard=keyboard)

        elif cmd == 'show_cars':
            cars = get_user_cars(uid)
            logger.info(f"show_cars: cars={cars}")
            if not cars:
                answer("У вас нет машин")
                return
            answer("Загружаю машины...")
            send(api, uid, "🚗 ВАШИ АВТОМОБИЛИ 🚗\n", peer_id)
            for car_name in cars:
                car_info = None
                for key, car in CARS.items():
                    if car['name'].lower() == car_name.lower():
                        car_info = car
                        break
                if car_info:
                    send(api, uid, f"{car_info['emoji']} {car_name}\n💰 Стоимость: {fmt_big(car_info['price'])}$", peer_id, attachment=car_info['photo'])
                else:
                    send(api, uid, f"🚗 {car_name}", peer_id)

        elif cmd in ('marry_accept', 'marry_decline'):
            proposer_id = payload.get('from')
            target_id   = payload.get('to')
            # Кнопку должен нажать именно адресат
            if uid != target_id:
                answer("Это предложение не для вас 😅")
                return
            proposer = get_user(proposer_id)
            target   = get_user(target_id)
            if not proposer or not target:
                answer("Ошибка: пользователь не найден")
                return
            # Очищаем предложения у обоих
            def clear_proposals(u_id, other_id):
                u = get_user(u_id)
                try:
                    props = json.loads(u['proposals_in'] or '[]')
                except:
                    props = []
                if other_id in props:
                    props.remove(other_id)
                with sqlite3.connect(DB_NAME) as c:
                    c.execute("UPDATE users SET proposals_in=? WHERE id=?", (json.dumps(props), u_id))
                    c.commit()

            if cmd == 'marry_accept':
                if proposer['married_to'] or target['married_to']:
                    answer("Кто-то из вас уже в браке 💔")
                    return
                now_str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
                with sqlite3.connect(DB_NAME) as c:
                    c.execute("UPDATE users SET married_to=?, married_at=? WHERE id=?",
                              (target_id, now_str, proposer_id))
                    c.execute("UPDATE users SET married_to=?, married_at=? WHERE id=?",
                              (proposer_id, now_str, target_id))
                    c.commit()
                clear_proposals(proposer_id, target_id)
                clear_proposals(target_id, proposer_id)
                proposer_name, _ = get_vk_link(api, proposer_id)
                target_name, _   = get_vk_link(api, target_id)
                answer("💍 Вы приняли предложение!")
                send(api, uid,
                     f"💒 СВАДЬБА!\n"
                     f"━━━━━━━━━━━━━━━\n"
                     f"💍 {proposer_name} и {target_name} теперь в браке!\n"
                     f"━━━━━━━━━━━━━━━\n"
                     f"💌 /мойбрак — информация о браке",
                     peer_id)
            else:
                clear_proposals(proposer_id, target_id)
                clear_proposals(target_id, proposer_id)
                proposer_name, _ = get_vk_link(api, proposer_id)
                target_name, _   = get_vk_link(api, target_id)
                answer("❌ Вы отказали")
                send(api, uid,
                     f"💔 {target_name} отказал(а) {proposer_name}.",
                     peer_id)

    except Exception as e:
        logger.error(f"handle_message_event EXCEPTION: {e}", exc_info=True)

def main():
    init_db()
    logger.info("🚀 Запуск бота...")
    vk_session = VkApi(token=VK_TOKEN, api_version="5.131")
    
    while True:
        try:
            vk = vk_session.get_api()
            logger.info("✅ Авторизация успешна!")
            try:
                vk.groups.setLongPollSettings(
                    group_id=GROUP_ID,
                    enabled=1,
                    message_new=1,
                    message_event=1
                )
                logger.info("✅ LongPoll settings updated (message_event=1)")
            except Exception as e:
                logger.warning(f"setLongPollSettings: {e}")
            longpoll = VkBotLongPoll(vk_session, GROUP_ID)
            logger.info("✅ Бот работает! Напишите /меню")
            
            for event in longpoll.listen():
                logger.info(f"EVENT type={event.type}")
                if event.type == VkBotEventType.MESSAGE_NEW:
                    handle_message(vk, event)
                elif event.type == VkBotEventType.MESSAGE_EVENT:
                    handle_message_event(vk, event)
        except Exception as e:
            logger.error(f"❌ Ошибка: {e}")
            time.sleep(5)
            vk_session = VkApi(token=VK_TOKEN, api_version="5.131")

if __name__ == "__main__":
    main()
